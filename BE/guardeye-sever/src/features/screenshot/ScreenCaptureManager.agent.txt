// agent/src/collectors/ScreenCaptureManager.ts
//
// =============================================================================
// SCREEN CAPTURE MANAGER — Chụp màn hình khi browser thay đổi tiêu đề cửa sổ
// =============================================================================
//
// Thiết kế theo khuyến cáo của senior developer:
//
//   ✅ DÙNG heuristic "title change" thay vì keylogger
//   ✅ Upload ảnh TRỰC TIẾP lên server qua multipart/form-data
//   ✅ KHÔNG nhét ảnh vào DataBuffer (tránh OOM - Out of Memory)
//   ✅ Dùng isCapturing flag để tránh spam khi chuyển tab liên tục
//   ✅ startCaptureSequence() là ASYNC, KHÔNG await → không block Main Loop
//
// Cách sử dụng trong Main Loop (index.ts):
//
//   const BROWSER_PROCESSES = ['chrome.exe', 'msedge.exe', 'firefox.exe', 'brave.exe'];
//
//   // Trong runTick():
//   const windowEvent = await windowMonitor.collect();
//   if (windowEvent !== null) {
//     dataBuffer.push(windowEvent);
//
//     const procName = windowEvent.processName.toLowerCase();
//     if (BROWSER_PROCESSES.includes(procName)) {
//       // KHÔNG AWAIT — để chạy ngầm, không block tick
//       screenCaptureManager.startCaptureSequence(windowEvent.title);
//     }
//   }
//
// Cài đặt dependency:
//   npm install screenshot-desktop form-data node-fetch
//   npm install --save-dev @types/node-fetch @types/form-data
//
// =============================================================================

import screenshot from "screenshot-desktop";
import FormData from "form-data";
import https from "https";
import http from "http";

// -----------------------------------------------------------------------------
// CONSTANTS
// -----------------------------------------------------------------------------

const TOTAL_SHOTS = 3;          // Số ảnh mỗi batch
const CAPTURE_DURATION_MS = 10_000; // Tổng thời gian chụp: 10 giây
const INTERVAL_MS = Math.floor(CAPTURE_DURATION_MS / TOTAL_SHOTS); // ~3333ms/tấm

// Danh sách process trình duyệt được nhận diện
export const BROWSER_PROCESSES = [
  "chrome.exe",
  "msedge.exe",
  "firefox.exe",
  "brave.exe",
  "opera.exe",
  "vivaldi.exe",
];

// -----------------------------------------------------------------------------
// INTERFACE
// -----------------------------------------------------------------------------

interface ScreenCaptureManagerConfig {
  /** URL đầy đủ của endpoint upload, VD: "http://localhost:5000/api/v1/agent/screenshot" */
  serverUrl: string;
  /** Device token để xác thực với server */
  deviceToken: string;
}

// -----------------------------------------------------------------------------
// CLASS
// -----------------------------------------------------------------------------

export class ScreenCaptureManager {
  private isCapturing = false;
  private readonly serverUrl: string;
  private readonly deviceToken: string;

  constructor(config: ScreenCaptureManagerConfig) {
    this.serverUrl = config.serverUrl;
    this.deviceToken = config.deviceToken;
  }

  // ---------------------------------------------------------------------------
  // PUBLIC API
  // ---------------------------------------------------------------------------

  /**
   * Kích hoạt một chuỗi chụp màn hình (3 tấm trong 10 giây).
   *
   * QUAN TRỌNG: Gọi hàm này KHÔNG CẦN AWAIT.
   * Hàm chạy bất đồng bộ hoàn toàn — không block Main Loop.
   *
   * @param triggerTitle - Tiêu đề cửa sổ trình duyệt gây trigger
   */
  public startCaptureSequence(triggerTitle: string): void {
    // Guard: Tránh chạy nhiều chuỗi đồng thời khi user chuyển tab liên tục
    if (this.isCapturing) {
      console.log(
        `[ScreenCapture] Đang chụp — bỏ qua trigger mới: "${triggerTitle.slice(0, 50)}"`,
      );
      return;
    }

    this.isCapturing = true;
    console.log(
      `[ScreenCapture] Bắt đầu chụp do: "${triggerTitle.slice(0, 80)}"`,
    );

    // Chạy logic capture hoàn toàn async, không block caller
    this._runCaptureSequence(triggerTitle).catch((err) => {
      console.error("[ScreenCapture] Lỗi không xử lý được:", err);
      this.isCapturing = false;
    });
  }

  /**
   * Kiểm tra ScreenCaptureManager có đang trong giữa chuỗi chụp hay không.
   * Dùng để hiển thị status hoặc debug.
   */
  public get isBusy(): boolean {
    return this.isCapturing;
  }

  // ---------------------------------------------------------------------------
  // PRIVATE
  // ---------------------------------------------------------------------------

  /**
   * Logic nội bộ: chụp 3 tấm cách nhau INTERVAL_MS, upload từng tấm lên server.
   */
  private async _runCaptureSequence(triggerTitle: string): Promise<void> {
    const capturedAt = [
      new Date().toISOString(), // tấm 0: chụp ngay
      "",                        // tấm 1: sau INTERVAL_MS
      "",                        // tấm 2: sau 2*INTERVAL_MS
    ];

    for (let index = 0; index < TOTAL_SHOTS; index++) {
      if (index > 0) {
        // Chờ trước khi chụp tấm tiếp theo
        await this._sleep(INTERVAL_MS);
        capturedAt[index] = new Date().toISOString();
      }

      try {
        // Chụp màn hình — trả về Buffer (JPEG/PNG)
        const imgBuffer: Buffer = await screenshot({ format: "jpg" });

        console.log(
          `[ScreenCapture] Đã chụp tấm ${index + 1}/${TOTAL_SHOTS} (${imgBuffer.length} bytes)`,
        );

        // Upload ngay — không giữ nhiều buffer trong RAM cùng lúc
        await this._uploadScreenshot(
          imgBuffer,
          index as 0 | 1 | 2,
          capturedAt[index] || new Date().toISOString(),
          triggerTitle,
        );

        console.log(`[ScreenCapture] Đã upload tấm ${index + 1}/${TOTAL_SHOTS}`);
      } catch (err) {
        // Lỗi từng tấm → log và tiếp tục, không huỷ cả sequence
        console.error(
          `[ScreenCapture] Lỗi tấm ${index + 1}: ${(err as Error).message}`,
        );
      }
    }

    this.isCapturing = false;
    console.log("[ScreenCapture] Hoàn tất chuỗi chụp màn hình.");
  }

  /**
   * Upload 1 ảnh lên server qua multipart/form-data.
   *
   * API: POST /api/v1/agent/screenshot
   * Header: X-Device-Token: <deviceToken>
   * Body: multipart/form-data { screenshot: Buffer }
   * Query: ?captureIndex=<0|1|2>&capturedAt=<ISO>&triggerTitle=<string>
   */
  private _uploadScreenshot(
    imgBuffer: Buffer,
    captureIndex: 0 | 1 | 2,
    capturedAt: string,
    triggerTitle: string,
  ): Promise<void> {
    return new Promise((resolve, reject) => {
      // Tạo form data
      const form = new FormData();
      form.append("screenshot", imgBuffer, {
        filename: `screenshot-${Date.now()}.jpg`,
        contentType: "image/jpeg",
      });

      // Build URL với query params
      const queryParams = new URLSearchParams({
        captureIndex: String(captureIndex),
        capturedAt,
        triggerTitle: triggerTitle.slice(0, 500), // trim theo maxlength model
      });

      const fullUrl = `${this.serverUrl}?${queryParams.toString()}`;
      const url = new URL(fullUrl);

      // Headers = form headers + device token
      const formHeaders = form.getHeaders();
      const options = {
        hostname: url.hostname,
        port: url.port ? parseInt(url.port) : undefined,
        path: `${url.pathname}?${url.search.slice(1)}`,
        method: "POST",
        headers: {
          ...formHeaders,
          "X-Device-Token": this.deviceToken,
        },
      };

      // Chọn http hay https dựa theo protocol
      const transport = url.protocol === "https:" ? https : http;

      const req = transport.request(options, (res) => {
        let body = "";
        res.on("data", (chunk) => (body += chunk));
        res.on("end", () => {
          if (res.statusCode && res.statusCode >= 200 && res.statusCode < 300) {
            resolve();
          } else {
            reject(
              new Error(
                `Upload thất bại: HTTP ${res.statusCode} — ${body.slice(0, 200)}`,
              ),
            );
          }
        });
      });

      req.on("error", (err) => reject(err));

      // Set timeout 15 giây cho mỗi upload request
      req.setTimeout(15_000, () => {
        req.destroy(new Error("Upload timeout (15s)"));
      });

      form.pipe(req);
    });
  }

  /** Helper sleep */
  private _sleep(ms: number): Promise<void> {
    return new Promise((resolve) => setTimeout(resolve, ms));
  }
}
